# Banca ore (hour-bank) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Registro banca ore event-sourced per le risorse interne: l'OT si accumula, viene recuperato da assenze `recovery` (real-time) e giornate corte (a fine mese), con chiusura mensile obbligatoria che separa la quota pagata in busta da quella a recupero. Solo HR; cost report invariato.

**Architecture:** Ledger append-only `BancaOreEntry` (saldo = Σ ore firmate) + `BancaOreMarker` (chiusura mensile per risorsa). Servizio puro-su-DB `app/services/banca_ore.py` per accrual/deficit/drawdown/saldo/chiusura. `recovery_mode` configurabile su `WorkingHoursPolicy`. `recovery` smette di consumare ROL e attinge la banca. Endpoint read + azioni in `app/routers/hr.py`; hook all'approvazione recupero in `app/routers/planning_unavailabilities.py`.

**Tech Stack:** FastAPI, SQLAlchemy 2.0 (`Mapped`/`mapped_column`), Jinja2, vanilla JS, pytest (sqlite in-memory), Playwright (smoke). Spec: `docs/superpowers/specs/2026-06-16-banca-ore-design.md`.

"""Crea le tabelle banca ore e aggiunge le colonne WHP. Idempotente.

Uso: .venv\\Scripts\\python.exe scripts\\migrate_banca_ore.py
"""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parents[1]))

from sqlalchemy import inspect, text
from app.database import engine, Base
from app.models import models  # noqa: F401  (registra i modelli)


def _add_col(conn, table, col, ddl):
    insp = inspect(conn)
    cols = [c["name"] for c in insp.get_columns(table)]
    if col not in cols:
        conn.execute(text(f"ALTER TABLE {table} ADD COLUMN {ddl}"))
        print(f"  + {table}.{col}")


def main():
    # 1) crea tabelle mancanti (create_all è idempotente)
    Base.metadata.create_all(engine, tables=[
        models.BancaOreEntry.__table__, models.BancaOreMarker.__table__,
    ])
    print("Tabelle banca_ore_entries / banca_ore_markers OK.")
    # 2) colonne WHP
    with engine.begin() as conn:
        _add_col(conn, "working_hours_policies", "banca_ore_enabled",
                 "banca_ore_enabled BOOLEAN DEFAULT 0")
        _add_col(conn, "working_hours_policies", "recovery_mode",
                 "recovery_mode VARCHAR(40) DEFAULT 'ratio_1to1_pay_premium'")
    print("Migrazione banca ore completata.")


if __name__ == "__main__":
    main()
```

"""Servizio banca ore (hour-bank). Solo HR; nessun impatto cost report.

Ledger event-sourced: il saldo è la somma delle ore firmate di BancaOreEntry.
"""
from __future__ import annotations
from datetime import date, datetime
from typing import Optional

from sqlalchemy import func
from sqlalchemy.orm import Session

from app.models.models import (
    BancaOreEntry, BancaOreMarker, BancaOreEntryKind, MarkerStatus, RecoveryMode,
    Resource, ResourceType,
)

_INTERNAL = {ResourceType.person_internal, ResourceType.person}


def is_eligible(resource) -> bool:
    """Solo risorse interne (dipendenti) hanno banca ore."""
    return getattr(resource, "resource_type", None) in _INTERNAL


def balance(db: Session, resource_id: int, *, tenant_id: int = 1) -> float:
    """Saldo = Σ ore firmate delle entry (può essere negativo)."""
    total = (
        db.query(func.coalesce(func.sum(BancaOreEntry.hours), 0.0))
        .filter(BancaOreEntry.tenant_id == tenant_id,
                BancaOreEntry.resource_id == resource_id)
        .scalar()
    )
    return round(float(total or 0.0), 2)


def _add_entry(db: Session, *, resource_id, entry_date, hours, kind,
               source_type=None, source_id=None, note=None,
               created_by_user_id=None, tenant_id: int = 1) -> BancaOreEntry:
    e = BancaOreEntry(
        tenant_id=tenant_id, resource_id=resource_id, entry_date=entry_date,
        hours=round(float(hours), 2), kind=kind.value if hasattr(kind, "value") else kind,
        source_type=source_type, source_id=source_id,
        period_year=entry_date.year, period_month=entry_date.month,
        note=note, created_by_user_id=created_by_user_id,
    )
    db.add(e)
    db.flush()
    return e
```

def short_day_deficit(*, ordinary_hours: float, worked_hours: float,
                      justified_absence_hours: float) -> float:
    """Deficit di una giornata lavorativa schedulata: ore ordinarie non coperte.

    PURA. Il chiamante passa già un giorno lavorativo (no weekend/festivi) e le
    ore lavorate/assenze approvate di quel giorno. deficit = max(0, ord - lav - ass).
    """
    return round(max(0.0, float(ordinary_hours) - float(worked_hours)
                     - float(justified_absence_hours)), 2)

def bank_credit_and_premium(*, banked_base: float, mult: float,
                            mode: RecoveryMode) -> tuple[float, float]:
    """Dato il monte ore base a recupero e la maggiorazione media `mult`,
    ritorna (credito_banca, premio_pagato_ore) secondo il recovery_mode."""
    b = float(banked_base)
    m = float(mult)
    if mode == RecoveryMode.weighted_no_pay:
        return round(b * m, 2), 0.0
    if mode == RecoveryMode.flat_no_premium:
        return round(b, 2), 0.0
    # ratio_1to1_pay_premium (default)
    return round(b, 2), round(b * (m - 1.0), 2)
```

def _month_overtime(db: Session, resource, year: int, month: int) -> tuple[float, float]:
    """(ore OT base del mese, maggiorazione media). Riusa overtime.compute_overtime
    sui TimePunch del mese della risorsa. mult medio = weighted_factor/ore_OT se >0,
    altrimenti overtime_multiplier della policy (fallback 1.0)."""
    from app.services.overtime import compute_overtime
    from app.models.models import TimePunch
    from sqlalchemy import extract
    policy = _policy_for(db, resource)
    punches = (db.query(TimePunch)
               .filter(TimePunch.resource_id == resource.id,
                       extract("year", TimePunch.start_datetime) == year,
                       extract("month", TimePunch.start_datetime) == month)
               .all())
    br = compute_overtime(punches, policy)
    ot = br.overtime_daily_hours + br.overtime_weekly_hours
    if ot <= 0:
        return 0.0, 1.0
    # maggiorazione media = ore equivalenti OT / ore OT (cap minimo 1.0)
    mult = (br.weighted_factor / ot) if br.weighted_factor and ot else \
        (policy.overtime_multiplier or 1.0)
    return round(ot, 2), round(max(1.0, mult), 4)


def _policy_for(db: Session, resource) -> "WorkingHoursPolicy":
    """Policy applicabile alla risorsa (override risorsa→tenant default). Riusa la
    logica già esistente in overtime/hr per la selezione policy; fallback al primo
    WorkingHoursPolicy attivo del tenant."""

    pid = getattr(resource, "working_hours_policy_id", None)
    if pid:

def _worked_hours_on(db, resource_id, day) -> float:
    from app.models.models import TimePunch, TimePunchKind
    from datetime import datetime, time as _t
    start = datetime.combine(day, _t.min); end = datetime.combine(day, _t.max)
    punches = (db.query(TimePunch)
               .filter(TimePunch.resource_id == resource_id,
                       TimePunch.start_datetime >= start,
                       TimePunch.start_datetime <= end,
                       TimePunch.end_datetime.isnot(None)).all())

        tot += max(0.0, secs / 3600.0 - (getattr(p, "break_minutes", 0) or 0) / 60.0)
    return round(tot, 2)

def close_month(db: Session, *, resource, policy, year: int, month: int,
                cashout_hours: float, created_by_user_id: Optional[int] = None,
                tenant_id: int = 1) -> BancaOreMarker:
    if not is_eligible(resource):
        raise ValueError("Risorsa non idonea alla banca ore")
    existing = get_marker(db, resource.id, year, month, tenant_id=tenant_id)
    if existing and existing.status == MarkerStatus.closed.value:
        raise ValueError("Mese già chiuso: riaprire prima di ri-chiudere")
    ot_hours, mult = _month_overtime(db, resource, year, month)
    cashout = max(0.0, min(float(cashout_hours), ot_hours))
    banked_base = ot_hours - cashout
    credit, premium = bank_credit_and_premium(
        banked_base=banked_base, mult=mult,
        mode=RecoveryMode(policy.recovery_mode))
    eom = date(year, month, 28)  # data competenza accrual: fine mese
    # accrual OT
    if credit:
        _add_entry(db, resource_id=resource.id, entry_date=eom, hours=credit,
                   kind=BancaOreEntryKind.accrual_overtime,
                   source_type="banca_ore_marker", source_id=None,
                   created_by_user_id=created_by_user_id, tenant_id=tenant_id,
                   note=f"OT {year}-{month:02d} accreditato")
    # deficit giornate corte
    for day, dfc in _month_short_day_deficits(db, resource, policy, year, month):
        _add_entry(db, resource_id=resource.id, entry_date=day, hours=-dfc,
                   kind=BancaOreEntryKind.short_day_deficit,
                   source_type="banca_ore_marker",
                   created_by_user_id=created_by_user_id, tenant_id=tenant_id)
    marker = existing or BancaOreMarker(
        tenant_id=tenant_id, resource_id=resource.id,
        period_year=year, period_month=month)
    marker.computed_ot_hours = ot_hours
    marker.cashout_hours = cashout
    marker.banked_hours = credit
    marker.premium_paid = premium
    marker.status = MarkerStatus.closed.value
    marker.closed_at = datetime.utcnow()
    marker.closed_by_user_id = created_by_user_id
    db.add(marker); db.flush()
    return marker


def reopen_month(db: Session, *, resource_id: int, year: int, month: int,
                 created_by_user_id: Optional[int] = None, tenant_id: int = 1):
    """Storna le entry generate dalla chiusura (manual_adjustment di segno opposto)
    e rimette il marker open. NON cancella le entry (ledger immutabile)."""
    marker = get_marker(db, resource_id, year, month, tenant_id=tenant_id)
    if not marker or marker.status != MarkerStatus.closed.value:
        raise ValueError("Nessun mese chiuso da riaprire")
    gen = (db.query(BancaOreEntry)
           .filter(BancaOreEntry.tenant_id == tenant_id,
                   BancaOreEntry.resource_id == resource_id,
                   BancaOreEntry.source_type == "banca_ore_marker",
                   BancaOreEntry.period_year == year,
                   BancaOreEntry.period_month == month,
                   BancaOreEntry.kind.in_([
                       BancaOreEntryKind.accrual_overtime.value,
                       BancaOreEntryKind.short_day_deficit.value])).all())
    for e in gen:
        _add_entry(db, resource_id=resource_id, entry_date=e.entry_date,
                   hours=-e.hours, kind=BancaOreEntryKind.manual_adjustment,
                   source_type="banca_ore_marker_reopen", source_id=e.id,
                   created_by_user_id=created_by_user_id, tenant_id=tenant_id,
                   note=f"Storno riapertura {year}-{month:02d}")
    marker.status = MarkerStatus.open.value
    marker.closed_at = None
    db.add(marker); db.flush()
    return marker
```
Nota: lo storno somma entry opposte ma NON deduplica una doppia riapertura — guardia: `reopen_month` agisce solo se `status==closed`, e dopo lo storno il marker è `open`, quindi una seconda riapertura senza ri-chiusura solleva. Le entry originali + storni si annullano (saldo torna a includere solo gli altri movimenti). Per la ri-chiusura dopo riapertura, `close_month` ricrea nuove accrual/deficit: va bene perché lo storno ha neutralizzato le precedenti.
